## 二、高优先级风险点

### 2.1 一级风险（可能导致系统停机）

#### 1. 硬件通信无超时保护

```
AGVController.py: TCP连接无超时
ArmController.py: HTTP请求无超时
→ 某个硬件卡死会阻塞整个调度器
```

#### 2. gRPC连接断开无重连

```
RobotControlSystem.py: 流中断后无自动恢复
→ 需要外部重启才能恢复通信
```

#### 3. 任务执行无超时

```
TaskScheduler.py: 命令执行无超时限制
→ 一个任务卡死会阻塞整个队列
```

### 2.2 二级风险（影响可靠性）

- 队列深度无限制（内存风险）
- 数据库操作无超时（死锁风险）
- 线程异常可能被静默吞掉

### 2.3 风险汇总表

| 问题 | 影响范围 | 优先级 | 工作量 |
|------|---------|--------|--------|
| 硬件操作无超时保护 | 整个系统 | 🔴高 | 中 |
| 无硬件自动恢复机制 | AGV/机械臂 | 🔴高 | 大 |
| gRPC连接无重连机制 | 通信模块 | 🔴高 | 中 |
| 配置硬编码分散 | 运维维护 | 🟠中 | 小 |
| 缺少性能监控 | 运维监控 | 🟠中 | 中 |
| 队列无深度限制 | 内存管理 | 🟠中 | 小 |
| 数据库无连接池 | 数据访问 | 🟡低 | 中 |
| 告警机制缺失 | 运维告警 | 🟡低 | 大 |

建议创建 `core/timeout_utils.py`：

```python
import concurrent.futures
from functools import wraps

def with_timeout(timeout_seconds: float = 30):
    """装饰器：为任意函数添加超时保护"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as executor:
                future = executor.submit(func, *args, **kwargs)
                try:
                    return future.result(timeout=timeout_seconds)
                except concurrent.futures.TimeoutError:
                    raise TimeoutError(f"{func.__name__} 超时 ({timeout_seconds}s)")
        return wrapper
    return decorator

# 使用示例
@with_timeout(30)
def agv_moveto(self, marker):
    return self._send_command(...)
```

```python
from typing import Dict, Callable
from enum import Enum
import threading
import time

class HealthStatus(Enum):
    HEALTHY = "healthy"
    DEGRADED = "degraded"
    UNHEALTHY = "unhealthy"
    UNKNOWN = "unknown"

@dataclass
class ComponentHealth:
    name: str
    status: HealthStatus
    last_check: float
    error_count: int
    last_error: str = None

class HealthChecker:
    def __init__(self, check_interval: float = 30):
        self.components: Dict[str, Callable] = {}
        self.health_status: Dict[str, ComponentHealth] = {}
        self.check_interval = check_interval
        self._running = False

    def register(self, name: str, check_func: Callable[[], bool]):
        """注册组件健康检查函数"""
        self.components[name] = check_func
        self.health_status[name] = ComponentHealth(
            name=name, status=HealthStatus.UNKNOWN,
            last_check=0, error_count=0
        )

    def start(self):
        """启动定期健康检查"""
        self._running = True
        self._thread = threading.Thread(target=self._check_loop, daemon=True)
        self._thread.start()

    def get_system_health(self) -> Dict:
        """获取系统整体健康状态"""
        unhealthy = [h for h in self.health_status.values()
                     if h.status == HealthStatus.UNHEALTHY]
        return {
            "overall": HealthStatus.UNHEALTHY if unhealthy else HealthStatus.HEALTHY,
            "components": self.health_status,
            "unhealthy_count": len(unhealthy)
        }
```

### 3.3 安全控制器包装

建议创建 `core/safe_controller.py`：

```python
class SafeController:
    """为硬件控制器提供安全包装"""

    def __init__(self, controller, name: str, timeout: float = 30):
        self._controller = controller
        self.name = name
        self.timeout = timeout
        self.error_count = 0
        self.is_healthy = True
        self.logger = get_logger(f"SafeController.{name}")

    def safe_call(self, method_name: str, *args, **kwargs):
        """安全调用控制器方法"""
        if not self.is_healthy:
            self.logger.warning(f"{self.name} 不健康，跳过调用 {method_name}")
            return None

        method = getattr(self._controller, method_name)
        try:
            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as executor:
                future = executor.submit(method, *args, **kwargs)
                result = future.result(timeout=self.timeout)
                self.error_count = 0  # 成功后重置
                return result
        except TimeoutError:
            self.error_count += 1
            self.logger.error(f"{self.name}.{method_name} 超时")
            self._check_health()
            return None
        except Exception as e:
            self.error_count += 1
            self.logger.error(f"{self.name}.{method_name} 异常: {e}")
            self._check_health()
            return None

    def _check_health(self):
        if self.error_count >= 3:
            self.is_healthy = False
            self.logger.critical(f"{self.name} 标记为不健康")
```

```python
import yaml
from pathlib import Path
from typing import Any, Dict
import os

class ConfigManager:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._config = {}
        return cls._instance

    def load(self, config_path: str = "conf/config.yaml"):
        """加载配置文件，支持环境变量覆盖"""
        with open(config_path, 'r', encoding='utf-8') as f:
            self._config = yaml.safe_load(f)
        self._apply_env_overrides()
        self._validate()
        return self

    def _apply_env_overrides(self):
        """环境变量覆盖 (LUMI_ROBOT_ID, LUMI_GRPC_HOST等)"""
        env_mapping = {
            'LUMI_ROBOT_ID': ('robot', 'id'),
            'LUMI_GRPC_HOST': ('grpc', 'server_host'),
            'LUMI_GRPC_PORT': ('grpc', 'server_port'),
        }
        for env_var, path in env_mapping.items():
            if env_var in os.environ:
                self._set_nested(path, os.environ[env_var])

    def get(self, *keys, default=None) -> Any:
        """获取配置值 config.get('hardware', 'agv', 'timeout')"""
        value = self._config
        for key in keys:
            if isinstance(value, dict) and key in value:
                value = value[key]
            else:
                return default
        return value
```

### 4.2 gRPC自动重连机制

修改 `RobotControlSystem.py`，添加连接管理器：

```python
class GrpcConnectionManager:
    """gRPC连接管理器，支持自动重连"""

    def __init__(self, config: Dict):
        self.config = config
        self.channel = None
        self.stub = None
        self.is_connected = False
        self.reconnect_attempts = 0
        self.max_attempts = config.get('max_reconnect_attempts', 10)
        self.reconnect_interval = config.get('reconnect_interval', 5)
        self._reconnect_lock = threading.Lock()

    def connect(self) -> bool:
        """建立连接"""
        try:
            self.channel = grpc.insecure_channel(
                f"{self.config['server_host']}:{self.config['server_port']}",
                options=[
                    ('grpc.keepalive_time_ms', 60000),
                    ('grpc.keepalive_timeout_ms', 20000),
                ]
            )
            grpc.channel_ready_future(self.channel).result(
                timeout=self.config.get('connection_timeout', 10)
            )
            self.stub = RobotServiceStub(self.channel)
            self.is_connected = True
            self.reconnect_attempts = 0
            return True
        except Exception as e:
            self.logger.error(f"连接失败: {e}")
            return False

    def reconnect(self):
        """自动重连（指数退避）"""
        with self._reconnect_lock:
            while self.reconnect_attempts < self.max_attempts:
                self.reconnect_attempts += 1
                wait_time = min(self.reconnect_interval * (2 ** self.reconnect_attempts), 300)
                self.logger.info(f"尝试重连 ({self.reconnect_attempts}/{self.max_attempts})，等待 {wait_time}s")
                time.sleep(wait_time)

                if self.connect():
                    self.logger.info("重连成功")
                    return True

            self.logger.critical("重连失败，已达最大尝试次数")
            return False
```

```python
from dataclasses import dataclass, field
from typing import Dict
import threading
import time
import json

@dataclass
class SystemMetrics:
    """系统指标收集"""

    # 任务指标
    commands_queued: int = 0
    commands_completed: int = 0
    commands_failed: int = 0
    commands_timeout: int = 0

    # 硬件指标
    agv_errors: int = 0
    arm_errors: int = 0
    camera_errors: int = 0

    # 通信指标
    grpc_messages_sent: int = 0
    grpc_messages_received: int = 0
    grpc_reconnect_count: int = 0

    # 性能指标
    avg_command_duration_ms: float = 0
    queue_depth: int = 0

    # 时间戳
    last_update: float = field(default_factory=time.time)

    def to_dict(self) -> Dict:
        return {
            "commands": {
                "queued": self.commands_queued,
                "completed": self.commands_completed,
                "failed": self.commands_failed,
                "success_rate": self._calc_success_rate()
            },
            "hardware": {
                "agv_errors": self.agv_errors,
                "arm_errors": self.arm_errors,
                "camera_errors": self.camera_errors
            },
            "grpc": {
                "sent": self.grpc_messages_sent,
                "received": self.grpc_messages_received,
                "reconnects": self.grpc_reconnect_count
            },
            "performance": {
                "avg_duration_ms": self.avg_command_duration_ms,
                "queue_depth": self.queue_depth
            }
        }

    def _calc_success_rate(self) -> float:
        total = self.commands_completed + self.commands_failed
        return self.commands_completed / total if total > 0 else 1.0

class MetricsExporter:
    """指标导出器（支持Prometheus格式）"""

    def __init__(self, metrics: SystemMetrics, port: int = 9090):
        self.metrics = metrics
        self.port = port

    def export_prometheus(self) -> str:
        """导出Prometheus格式指标"""
        lines = []
        m = self.metrics

        lines.append(f'lumi_commands_total{{status="completed"}} {m.commands_completed}')
        lines.append(f'lumi_commands_total{{status="failed"}} {m.commands_failed}')
        lines.append(f'lumi_command_queue_depth {m.queue_depth}')
        lines.append(f'lumi_hardware_errors{{component="agv"}} {m.agv_errors}')
        lines.append(f'lumi_hardware_errors{{component="arm"}} {m.arm_errors}')
        lines.append(f'lumi_grpc_reconnects_total {m.grpc_reconnect_count}')

        return '\n'.join(lines)
```

```python
#!/usr/bin/env python3
"""健康检查脚本 - 可用于系统监控和告警"""

import socket
import sqlite3
import sys
from datetime import datetime, timedelta

def check_grpc_port(host='localhost', port=50051) -> bool:
    """检查gRPC端口是否可达"""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(5)
        result = sock.connect_ex((host, port))
        sock.close()
        return result == 0
    except:
        return False

def check_database() -> dict:
    """检查数据库状态"""
    try:
        conn = sqlite3.connect('tasks.db')
        cursor = conn.cursor()

        # 最近1小时的命令统计
        one_hour_ago = (datetime.now() - timedelta(hours=1)).isoformat()
        cursor.execute("""
            SELECT status, COUNT(*) FROM unified_commands
            WHERE created_at > ? GROUP BY status
        """, (one_hour_ago,))

        stats = dict(cursor.fetchall())
        conn.close()
        return {"ok": True, "stats": stats}
    except Exception as e:
        return {"ok": False, "error": str(e)}

def check_log_errors() -> dict:
    """检查最近的错误日志"""
    import os
    from collections import Counter

    log_dir = "logs"
    error_count = 0
    recent_errors = []

    if not os.path.exists(log_dir):
        return {"error_count": 0, "recent_errors": []}

    for log_file in os.listdir(log_dir):
        if log_file.endswith('.log'):
            with open(os.path.join(log_dir, log_file), 'r', encoding='utf-8') as f:
                for line in f:
                    if 'ERROR' in line or 'CRITICAL' in line:
                        error_count += 1
                        recent_errors.append(line.strip()[:200])

    return {
        "error_count": error_count,
        "recent_errors": recent_errors[-10:]  # 最近10条
    }

def main():
    print("=== Lumi 健康检查 ===\n")

    # 1. gRPC连接检查
    grpc_ok = check_grpc_port()
    print(f"[{'✓' if grpc_ok else '✗'}] gRPC服务: {'正常' if grpc_ok else '不可达'}")

    # 2. 数据库检查
    db_result = check_database()
    print(f"[{'✓' if db_result['ok'] else '✗'}] 数据库: {'正常' if db_result['ok'] else db_result.get('error')}")
    if db_result['ok']:
        print(f"    最近1小时命令统计: {db_result['stats']}")

    # 3. 日志错误检查
    log_result = check_log_errors()
    print(f"[{'✓' if log_result['error_count'] < 10 else '⚠'}] 日志错误: {log_result['error_count']} 条")

    # 返回状态码
    if not grpc_ok or not db_result['ok']:
        sys.exit(1)
    sys.exit(0)

if __name__ == '__main__':
    main()
```
