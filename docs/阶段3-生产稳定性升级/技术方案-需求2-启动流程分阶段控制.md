# 技术方案：需求2 - 启动流程分阶段控制

## 1. 需求回顾

**目标**：将系统启动分为3个独立阶段，实现远程控制硬件启停

| 阶段 | 触发条件 | 行为 |
|------|----------|------|
| 阶段1 | 程序启动 | 仅启动gRPC通信，等待指令 |
| 阶段2 | 收到 `HARDWARE_START_CMD` | 启动硬件模块 |
| 阶段3 | 收到任务指令 | 执行任务 |

**验收标准**：
- 程序启动后，gRPC连接成功，但硬件未启动
- 可以与服务端正常通信
- 硬件模块必须等待服务端命令才能启动

---

## 2. 硬件模块划分

| 模块组 | 包含组件 | 控制方式 |
|--------|----------|----------|
| **robot** | AGV + 机械臂 + 外部轴 | 整体控制 |
| **camera** | 深度相机 | 单独控制 |
| **env_sensor** | 环境传感器 | 单独控制 |

**状态定义**：仅 `enable` 和 `disable` 两种状态

---

## 3. 配置项

在 `conf/config.json` 中新增 `hardware_config` 配置块：

```json
{
    "robot_id": 12345,
    "use_mock": true,
    "hardware_config": {
        "auto_start_on_boot": false,  // 程序启动时是否自动启动硬件
        "auto_start_robot": true,     // 自动启动时是否启动机器人
        "auto_start_camera": true,    // 自动启动时是否启动相机
        "auto_start_env_sensor": true // 自动启动时是否启动环境传感器
    },
    "grpc_config": { ... },
    "robot_config": { ... }
}
```

| 配置项 | 类型 | 默认值 | 说明 |
|--------|------|--------|------|
| `auto_start_on_boot` | bool | `false` | 程序启动时是否自动启动硬件 |
| `auto_start_robot` | bool | `true` | 自动启动时是否启动机器人（AGV+机械臂+外部轴） |
| `auto_start_camera` | bool | `true` | 自动启动时是否启动相机 |
| `auto_start_env_sensor` | bool | `true` | 自动启动时是否启动环境传感器 |

**说明**：
- `auto_start_on_boot=false` 时，程序启动后仅建立gRPC通信，等待服务端下发硬件启动命令
- `auto_start_on_boot=true` 时，程序启动后自动启动配置的硬件模块（兼容旧行为）

### 5.2 TaskManager.py

```python
class TaskManager:
    def __init__(self, config: Dict[str, Any] = None, use_mock: bool = True,
                 auto_start_hardware: bool = True):
        """
        Args:
            auto_start_hardware: 是否自动启动硬件（默认True保持向后兼容）
        """
        self.config = config or {}
        self.use_mock = use_mock
        self.logger = get_logger(__name__)

        # 硬件状态（enable/disable）
        self._hardware_status = {
            "robot": False,       # AGV + 机械臂 + 外部轴
            "camera": False,
            "env_sensor": False
        }
        self._hardware_lock = threading.Lock()

        # 初始化数据库
        self.database = TaskDatabase()

        # 创建机器人控制器（但不启动）
        self._create_robot_controller()

        # 根据参数决定是否自动启动硬件
        if auto_start_hardware:
            self.start_hardware(robot=True, camera=True, env_sensor=True)

        # 初始化调度器
        self.scheduler = TaskScheduler(self.robot_controller, self.database)
        if auto_start_hardware:
            self.scheduler.start()

        # 注册回调
        self._register_scheduler_callbacks()

        # 系统级回调
        self.system_callbacks = {
            "on_command_status_change": None,
            "on_task_progress": None,
            "on_operation_result": None,
        }

    def _create_robot_controller(self):
        """创建机器人控制器（不启动硬件）"""
        if self.use_mock:
            from robot.MockRobotController import MockRobotController
            self.robot_controller = MockRobotController(
                self.config.get('robot_config', {}),
                auto_setup=False
            )
        else:
            from robot.RobotController import RobotController
            self.robot_controller = RobotController(
                self.config,
                auto_setup=False
            )
        self.logger.info("机器人控制器已创建（未启动）")

    def start_hardware(self, robot: bool = False, camera: bool = False,
                       env_sensor: bool = False) -> dict:
        """启动指定硬件模块

        Args:
            robot: 是否启动机器人（AGV+机械臂+外部轴）
            camera: 是否启动相机
            env_sensor: 是否启动环境传感器

        Returns:
            dict: 各模块启动结果
        """
        result = {"robot": False, "camera": False, "env_sensor": False}

        with self._hardware_lock:
            # 启动机器人模块
            if robot and not self._hardware_status["robot"]:
                try:
                    if self.robot_controller.setup_system():
                        self._hardware_status["robot"] = True
                        result["robot"] = True
                        # 启动调度器
                        if not self.scheduler._is_running:
                            self.scheduler.start()
                        self.logger.info("机器人模块已启动")
                except Exception as e:
                    self.logger.error(f"机器人模块启动失败: {e}")

            # 启动相机
            if camera and not self._hardware_status["camera"]:
                try:
                    if self.robot_controller.start_camera():
                        self._hardware_status["camera"] = True
                        result["camera"] = True
                        self.logger.info("相机已启动")
                except Exception as e:
                    self.logger.error(f"相机启动失败: {e}")

            # 启动环境传感器
            if env_sensor and not self._hardware_status["env_sensor"]:
                try:
                    if self.robot_controller.start_env_sensor():
                        self._hardware_status["env_sensor"] = True
                        result["env_sensor"] = True
                        self.logger.info("环境传感器已启动")
                except Exception as e:
                    self.logger.error(f"环境传感器启动失败: {e}")

        return result

    def stop_hardware(self, robot: bool = False, camera: bool = False,
                      env_sensor: bool = False) -> dict:
        """关闭指定硬件模块

        Returns:
            dict: 各模块关闭结果
        """
        result = {"robot": False, "camera": False, "env_sensor": False}

        with self._hardware_lock:
            # 关闭机器人模块
            if robot and self._hardware_status["robot"]:
                try:
                    self.scheduler.stop()
                    self.robot_controller.shutdown_system()
                    self._hardware_status["robot"] = False
                    result["robot"] = True
                    self.logger.info("机器人模块已关闭")
                except Exception as e:
                    self.logger.error(f"机器人模块关闭失败: {e}")

            # 关闭相机
            if camera and self._hardware_status["camera"]:
                try:
                    self.robot_controller.stop_camera()
                    self._hardware_status["camera"] = False
                    result["camera"] = True
                    self.logger.info("相机已关闭")
                except Exception as e:
                    self.logger.error(f"相机关闭失败: {e}")

            # 关闭环境传感器
            if env_sensor and self._hardware_status["env_sensor"]:
                try:
                    self.robot_controller.stop_env_sensor()
                    self._hardware_status["env_sensor"] = False
                    result["env_sensor"] = True
                    self.logger.info("环境传感器已关闭")
                except Exception as e:
                    self.logger.error(f"环境传感器关闭失败: {e}")

        return result

    def get_hardware_status(self) -> dict:
        """获取硬件状态"""
        with self._hardware_lock:
            return self._hardware_status.copy()

    def is_robot_ready(self) -> bool:
        """检查机器人是否就绪（可执行任务）"""
        with self._hardware_lock:
            return self._hardware_status["robot"]
```

### 5.3 RobotControlSystem.py

```python
class RobotControlSystem:
    def __init__(self, config, use_mock=True, report=True):
        # ... 现有初始化代码 ...
        self.task_manager = None

        # 读取硬件配置
        self.hardware_config = self.config.get('hardware_config', {})
        self.auto_start_on_boot = self.hardware_config.get('auto_start_on_boot', False)

    def start(self):
        """启动系统"""
        if self.is_running:
            return

        self.logger.info("启动机器人控制系统...")

        # 根据配置决定是否自动启动硬件
        self.task_manager = TaskManager(
            self.config,
            use_mock=self.use_mock,
            auto_start_hardware=self.auto_start_on_boot
        )

        # 如果配置为自动启动，根据子配置启动指定模块
        if self.auto_start_on_boot:
            self.task_manager.start_hardware(
                robot=self.hardware_config.get('auto_start_robot', True),
                camera=self.hardware_config.get('auto_start_camera', True),
                env_sensor=self.hardware_config.get('auto_start_env_sensor', True)
            )
            self.logger.info("硬件已自动启动")
        else:
            self.logger.info("等待服务端下发硬件启动指令")

        # 注册回调
        self._register_task_manager_callbacks()

        # 启动gRPC通信
        self._start_communication()

        self.is_running = True

    def _start_communication(self):
        """启动gRPC通信"""
        self._shutdown_event.clear()
        self._reconnect_trigger.clear()
        self._start_heartbeat_monitor()
        self._start_reconnect_manager()

        self._set_connection_state(ConnectionState.CONNECTING)
        if self._init_grpc_client():
            self.set_client_upload_response_handler(self._handle_clientUpload_response)
            self.set_server_command_response_handler(self._handle_serverCommand)

        if self.report:
            self._start_reporting()

    def _handle_serverCommand(self, response):
        """处理服务端命令"""
        command_envelope = convert_server_message_to_command_envelope(response)
        cmd_type = command_envelope.cmd_type

        # 硬件启动命令
        if cmd_type == CmdType.HARDWARE_START_CMD:
            self._handle_hardware_start(command_envelope)
            return

        # 硬件关闭命令
        if cmd_type == CmdType.HARDWARE_SHUTDOWN_CMD:
            self._handle_hardware_shutdown(command_envelope)
            return

        # 其他命令走原有流程（上位机负责控制时序）
        # ... 现有逻辑 ...

    def _handle_hardware_start(self, command_envelope):
        """处理硬件启动命令"""
        data = command_envelope.data_json.get('hardware_control_cmd', {})

        result = self.task_manager.start_hardware(
            robot=data.get('robot', False),
            camera=data.get('camera', False),
            env_sensor=data.get('env_sensor', False)
        )

        self._send_hardware_status_response(command_envelope.cmd_id)
        self.logger.info(f"硬件启动结果: {result}")

    def _handle_hardware_shutdown(self, command_envelope):
        """处理硬件关闭命令"""
        data = command_envelope.data_json.get('hardware_control_cmd', {})

        result = self.task_manager.stop_hardware(
            robot=data.get('robot', False),
            camera=data.get('camera', False),
            env_sensor=data.get('env_sensor', False)
        )

        self._send_hardware_status_response(command_envelope.cmd_id)
        self.logger.info(f"硬件关闭结果: {result}")

    def _send_hardware_status_response(self, command_id: str):
        """发送硬件状态响应"""
        status = self.task_manager.get_hardware_status()

        client_msg = robot_pb2.ClientStreamMessage(
            command_id=int(command_id) if command_id.isdigit() else 0,
            command_time=int(time.time() * 1000),
            command_type=robot_pb2.ClientMessageType.HARDWARE_STATUS_RESPONSE,
            robot_id=self.robot_id,
            hardware_status=robot_pb2.HardwareStatusResponse(
                robot_enabled=status["robot"],
                camera_enabled=status["camera"],
                env_sensor_enabled=status["env_sensor"],
                message="OK"
            )
        )

        if self.server_command_manager:
            self.server_command_manager.send_message(client_msg)
```

## 6. 启动流程

### 配置 `auto_start_on_boot: false`（默认）

```
程序启动
    │
    ▼
RobotControlSystem.start()
    │
    ├──► TaskManager(auto_start_hardware=False)
    │         │
    │         └──► 创建控制器（不启动硬件）
    │
    ▼
启动gRPC通信 ──► 阶段1: 通信就绪
    │
    ▼
等待服务端指令
    │
    ├──► HARDWARE_START_CMD {robot: true}
    │         │
    │         └──► 启动 AGV + 机械臂 + 外部轴
    │
    ├──► HARDWARE_START_CMD {camera: true}
    │         │
    │         └──► 启动相机
    │
    └──► TASK_CMD
              │
              └──► 执行任务
```

### 配置 `auto_start_on_boot: true`（兼容旧行为）

```
程序启动
    │
    ▼
RobotControlSystem.start()
    │
    ├──► TaskManager(auto_start_hardware=True)
    │         │
    │         └──► 根据配置自动启动硬件
    │

启动gRPC通信 ──► 硬件已就绪

    ▼
```
