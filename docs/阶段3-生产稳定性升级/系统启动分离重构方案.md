# 系统启动分离重构方案

## 1. 背景与目标

### 1.1 当前架构问题

当前 `RobotControlSystem.start()` 方法中，gRPC 通信和机器人控制器同时初始化：

```python
def start(self):
    # 同时初始化 TaskManager（内部会初始化 RobotController）
    self.task_manager = TaskManager(self.config, use_mock=self.use_mock)

    # 初始化 gRPC 客户端
    if not self._init_grpc_client():
        ...
```

问题：
1. 机器人硬件在系统启动时就初始化，无法由上位机控制
2. 无法单独初始化/使能/关闭某个硬件组件（如只初始化 AGV 或只初始化机械臂）
3. gRPC 连接和硬件初始化强耦合

### 1.2 目标架构

```
┌─────────────────────────────────────────────────────────────────┐
│                    RobotControlSystem                            │
├─────────────────────────────────────────────────────────────────┤
│  ┌───────────────────┐      ┌─────────────────────────────────┐ │
│  │  gRPC 通信层       │      │     硬件控制层                   │ │
│  │  (系统启动时初始化) │ ───→ │ (收到系统命令后按需初始化)        │ │
│  │                   │      │                                 │ │
│  │ - StreamManager   │      │  ┌─────────────────────────┐   │ │
│  │ - 命令接收/状态上报 │      │  │   HardwareManager       │   │ │
│  └───────────────────┘      │  │   (新增组件管理器)        │   │ │
│                              │  │   - AGV                 │   │ │
│                              │  │   - Arm (机械臂)         │   │ │
│                              │  │   - ExtAxis (外部轴)     │   │ │
│                              │  │   - Camera (相机)        │   │ │
│                              │  │   - EnvSensor (环境传感器)│   │ │
│                              │  └─────────────────────────┘   │ │
│                              │                                 │ │
│                              │  ┌─────────────────────────┐   │ │
│                              │  │   TaskManager           │   │ │
│                              │  │   (任务管理，延迟初始化)  │   │ │
│                              │  └─────────────────────────┘   │ │
│                              └─────────────────────────────────┘ │
└─────────────────────────────────────────────────────────────────┘
```

## 2. 详细设计

### 2.1 新增命令类型

在 `dataModels/CommandModels.py` 中新增系统控制命令类型：

```python
class CmdType(Enum):

    # 新增：系统控制命令
    SYSTEM_CONTROL_CMD = "system_control_cmd"
```

新增系统控制命令数据模型：

```python
class SystemControlAction(Enum):
    """系统控制动作"""
    INIT = "init"           # 初始化
    ENABLE = "enable"       # 使能
    DISABLE = "disable"     # 禁用
    SHUTDOWN = "shutdown"   # 关闭

class HardwareComponent(Enum):
    """硬件组件类型"""
    ALL = "all"             # 所有组件
    AGV = "agv"             # AGV
    ARM = "arm"             # 机械臂
    EXT_AXIS = "ext_axis"   # 外部轴
    CAMERA = "camera"       # 相机
    ENV_SENSOR = "env_sensor"  # 环境传感器

@dataclass
class SystemControlCmd:
    """系统控制命令"""
    action: SystemControlAction      # 控制动作
    component: HardwareComponent     # 目标组件
    params: Dict[str, Any] = None    # 可选参数
```

### 2.2 新增 HardwareManager 组件管理器

创建 `robot/HardwareManager.py`：

```python
"""
HardwareManager.py
硬件组件管理器，支持各组件的独立初始化、使能、关闭
"""

from enum import Enum
from typing import Dict, Any, Optional
from utils.logger_config import get_logger
import threading

class ComponentState(Enum):
    """组件状态"""
    NOT_INITIALIZED = "not_initialized"  # 未初始化
    INITIALIZING = "initializing"        # 初始化中
    INITIALIZED = "initialized"          # 已初始化（但未使能）
    ENABLED = "enabled"                  # 已使能（可工作）
    DISABLED = "disabled"                # 已禁用
    ERROR = "error"                      # 错误状态

class HardwareManager:
    """硬件组件管理器"""

    def __init__(self, config: Dict[str, Any] = None):
        self.config = config or {}
        self.logger = get_logger("HardwareManager")

        # 组件状态
        self._component_states: Dict[str, ComponentState] = {
            "agv": ComponentState.NOT_INITIALIZED,
            "arm": ComponentState.NOT_INITIALIZED,
            "ext_axis": ComponentState.NOT_INITIALIZED,
            "camera": ComponentState.NOT_INITIALIZED,
            "env_sensor": ComponentState.NOT_INITIALIZED,
        }

        # 组件实例（延迟初始化）
        self._agv_controller = None
        self._arm_controller = None
        self._camera_manager = None
        self._env_sensor = None

        # 线程锁
        self._lock = threading.Lock()

        # 回调函数
        self.callbacks = {
            "on_component_state_change": [],
        }

    # ==================== 组件初始化方法 ====================

    def init_component(self, component: str, params: Dict = None) -> bool:
        """
        初始化指定组件

        Args:
            component: 组件名称 (agv, arm, ext_axis, camera, env_sensor, all)
            params: 初始化参数

        Returns:
            bool: 初始化是否成功
        """
        if component == "all":
            return self._init_all_components()

        init_methods = {
            "agv": self._init_agv,
            "arm": self._init_arm,
            "ext_axis": self._init_ext_axis,
            "camera": self._init_camera,
            "env_sensor": self._init_env_sensor,
        }

        if component not in init_methods:
            self.logger.error(f"未知组件: {component}")
            return False

        return init_methods[component](params)

    def _init_agv(self, params: Dict = None) -> bool:
        """初始化 AGV"""
        try:
            with self._lock:
                if self._component_states["agv"] != ComponentState.NOT_INITIALIZED:
                    self.logger.warning("AGV 已经初始化")
                    return True

                self._set_state("agv", ComponentState.INITIALIZING)

            self.logger.info("正在初始化 AGV 控制器...")

            from robot.AGVController import AGVController
            robot_config = self.config.get('robot_config', {})
            self._agv_controller = AGVController(robot_config)

            with self._lock:
                self._set_state("agv", ComponentState.INITIALIZED)

            self.logger.info("AGV 控制器初始化成功")
            return True

        except Exception as e:
            self.logger.error(f"AGV 初始化失败: {e}")
            self._set_state("agv", ComponentState.ERROR)
            return False

    def _init_arm(self, params: Dict = None) -> bool:
        """初始化机械臂（包含外部轴）"""
        try:
            with self._lock:
                if self._component_states["arm"] != ComponentState.NOT_INITIALIZED:
                    self.logger.warning("机械臂已经初始化")
                    return True

                self._set_state("arm", ComponentState.INITIALIZING)

            self.logger.info("正在初始化机械臂控制器...")

            from robot.ArmController import ArmController
            robot_config = self.config.get('robot_config', {})
            self._arm_controller = ArmController(
                system_config=robot_config,
                ext_axis_limits=robot_config.get("ext_axis_limits")
            )

            # 机械臂系统初始化
            if not self._arm_controller.setup_system():
                raise Exception("机械臂系统初始化失败")

            with self._lock:
                self._set_state("arm", ComponentState.INITIALIZED)
                self._set_state("ext_axis", ComponentState.INITIALIZED)

            self.logger.info("机械臂控制器初始化成功")
            return True

        except Exception as e:
            self.logger.error(f"机械臂初始化失败: {e}")
            self._set_state("arm", ComponentState.ERROR)
            return False

    def _init_ext_axis(self, params: Dict = None) -> bool:
        """初始化外部轴（通常与机械臂一起初始化）"""
        # 外部轴通过 ArmController 管理
        if self._arm_controller is None:
            self.logger.info("外部轴需要先初始化机械臂")
            return self._init_arm(params)
        return True

    def _init_camera(self, params: Dict = None) -> bool:
        """初始化相机"""
        try:
            camera_config = self.config.get('camera_config', {})
            if not camera_config.get('camera_enabled', False):
                self.logger.info("相机未启用")
                return True

            with self._lock:
                if self._component_states["camera"] != ComponentState.NOT_INITIALIZED:
                    self.logger.warning("相机已经初始化")
                    return True

                self._set_state("camera", ComponentState.INITIALIZING)

            self.logger.info("正在初始化相机...")

            from camera.CameraManager import CameraManager
            self._camera_manager = CameraManager(camera_config)

            if not self._camera_manager.start():
                raise Exception("相机启动失败")

            with self._lock:
                self._set_state("camera", ComponentState.INITIALIZED)

            self.logger.info("相机初始化成功")
            return True

        except Exception as e:
            self.logger.error(f"相机初始化失败: {e}")
            self._set_state("camera", ComponentState.ERROR)
            return False

    def _init_env_sensor(self, params: Dict = None) -> bool:
        """初始化环境传感器"""
        try:
            if not self.config.get('env_sensor_enabled', False):
                self.logger.info("环境传感器未启用")
                return True

            with self._lock:
                if self._component_states["env_sensor"] != ComponentState.NOT_INITIALIZED:
                    self.logger.warning("环境传感器已经初始化")
                    return True

                self._set_state("env_sensor", ComponentState.INITIALIZING)

            self.logger.info("正在初始化环境传感器...")

            from envsMonitor.AirQualitySensor import AirQualitySensor
            env_config = self.config.get('env_sensor_config', {})
            self._env_sensor = AirQualitySensor(
                port=env_config.get('port', 'COM4'),
                baudrate=env_config.get('baudrate', 4800),
                address=env_config.get('address', 0x01)
            )

            if not self._env_sensor.connect():
                raise Exception("环境传感器连接失败")

            with self._lock:
                self._set_state("env_sensor", ComponentState.INITIALIZED)

            self.logger.info("环境传感器初始化成功")
            return True

        except Exception as e:
            self.logger.error(f"环境传感器初始化失败: {e}")
            self._set_state("env_sensor", ComponentState.ERROR)
            return False

    def _init_all_components(self) -> bool:
        """初始化所有组件"""
        results = {
            "agv": self._init_agv(),
            "arm": self._init_arm(),
            "camera": self._init_camera(),
            "env_sensor": self._init_env_sensor(),
        }

        success_count = sum(1 for r in results.values() if r)
        total_count = len(results)

        self.logger.info(f"组件初始化完成: {success_count}/{total_count}")
        return all(results.values())

    # ==================== 组件使能/禁用方法 ====================

    def enable_component(self, component: str) -> bool:
        """使能组件"""
        if component == "all":
            return all([
                self._enable_single_component(c)
                for c in ["agv", "arm", "camera", "env_sensor"]
            ])
        return self._enable_single_component(component)

    def disable_component(self, component: str) -> bool:
        """禁用组件"""
        if component == "all":
            return all([
                self._disable_single_component(c)
                for c in ["agv", "arm", "camera", "env_sensor"]
            ])
        return self._disable_single_component(component)

    def _enable_single_component(self, component: str) -> bool:
        """使能单个组件"""
        with self._lock:
            current_state = self._component_states.get(component)
            if current_state == ComponentState.NOT_INITIALIZED:
                self.logger.error(f"组件 {component} 未初始化，无法使能")
                return False

            if current_state == ComponentState.ENABLED:
                self.logger.info(f"组件 {component} 已经使能")
                return True

            self._set_state(component, ComponentState.ENABLED)
            self.logger.info(f"组件 {component} 已使能")
            return True

    def _disable_single_component(self, component: str) -> bool:
        """禁用单个组件"""
        with self._lock:
            current_state = self._component_states.get(component)
            if current_state == ComponentState.NOT_INITIALIZED:
                self.logger.warning(f"组件 {component} 未初始化")
                return True

            self._set_state(component, ComponentState.DISABLED)
            self.logger.info(f"组件 {component} 已禁用")
            return True

    # ==================== 组件关闭方法 ====================

    def shutdown_component(self, component: str) -> bool:
        """关闭组件"""
        if component == "all":
            return self._shutdown_all_components()

        shutdown_methods = {
            "agv": self._shutdown_agv,
            "arm": self._shutdown_arm,
            "ext_axis": self._shutdown_ext_axis,
            "camera": self._shutdown_camera,
            "env_sensor": self._shutdown_env_sensor,
        }

        if component not in shutdown_methods:
            self.logger.error(f"未知组件: {component}")
            return False

        return shutdown_methods[component]()

    def _shutdown_agv(self) -> bool:
        """关闭 AGV"""
        try:
            if self._agv_controller:
                # AGV 控制器关闭逻辑
                self._agv_controller = None

            self._set_state("agv", ComponentState.NOT_INITIALIZED)
            self.logger.info("AGV 已关闭")
            return True
        except Exception as e:
            self.logger.error(f"关闭 AGV 失败: {e}")
            return False

    def _shutdown_arm(self) -> bool:
        """关闭机械臂"""
        try:
            if self._arm_controller:
                if hasattr(self._arm_controller, 'shutdown_system'):
                    self._arm_controller.shutdown_system()
                self._arm_controller = None

            self._set_state("arm", ComponentState.NOT_INITIALIZED)
            self._set_state("ext_axis", ComponentState.NOT_INITIALIZED)
            self.logger.info("机械臂已关闭")
            return True
        except Exception as e:
            self.logger.error(f"关闭机械臂失败: {e}")
            return False

    def _shutdown_ext_axis(self) -> bool:
        """关闭外部轴（通常与机械臂一起关闭）"""
        return self._shutdown_arm()

    def _shutdown_camera(self) -> bool:
        """关闭相机"""
        try:
            if self._camera_manager:
                self._camera_manager.stop()
                self._camera_manager = None

            self._set_state("camera", ComponentState.NOT_INITIALIZED)
            self.logger.info("相机已关闭")
            return True
        except Exception as e:
            self.logger.error(f"关闭相机失败: {e}")
            return False

    def _shutdown_env_sensor(self) -> bool:
        """关闭环境传感器"""
        try:
            if self._env_sensor:
                self._env_sensor.disconnect()
                self._env_sensor = None

            self._set_state("env_sensor", ComponentState.NOT_INITIALIZED)
            self.logger.info("环境传感器已关闭")
            return True
        except Exception as e:
            self.logger.error(f"关闭环境传感器失败: {e}")
            return False

    def _shutdown_all_components(self) -> bool:
        """关闭所有组件"""
        results = [
            self._shutdown_camera(),
            self._shutdown_env_sensor(),
            self._shutdown_arm(),
            self._shutdown_agv(),
        ]
        return all(results)

    # ==================== 状态查询方法 ====================

    def get_component_state(self, component: str) -> ComponentState:
        """获取组件状态"""
        return self._component_states.get(component, ComponentState.NOT_INITIALIZED)

    def get_all_states(self) -> Dict[str, str]:
        """获取所有组件状态"""
        return {k: v.value for k, v in self._component_states.items()}

    def is_component_ready(self, component: str) -> bool:
        """检查组件是否就绪（已初始化且已使能）"""
        state = self._component_states.get(component)
        return state == ComponentState.ENABLED

    def is_all_ready(self) -> bool:
        """检查所有必要组件是否就绪"""
        required = ["agv", "arm"]  # 必要组件
        return all(self.is_component_ready(c) for c in required)

    # ==================== 组件访问器 ====================

    @property
    def agv_controller(self):
        return self._agv_controller

    @property
    def arm_controller(self):
        return self._arm_controller

    @property
    def camera_manager(self):
        return self._camera_manager

    @property
    def env_sensor(self):
        return self._env_sensor

    # ==================== 内部方法 ====================

    def _set_state(self, component: str, state: ComponentState):
        """设置组件状态并触发回调"""
        old_state = self._component_states.get(component)
        self._component_states[component] = state

        if old_state != state:
            self._trigger_callback("on_component_state_change",
                                   component=component,
                                   old_state=old_state,
                                   new_state=state)

    def _trigger_callback(self, event: str, **kwargs):
        """触发回调"""
        for callback in self.callbacks.get(event, []):
            try:
                callback(**kwargs)
            except Exception as e:
                self.logger.error(f"回调执行失败: {e}")
```

### 2.3 修改 RobotControlSystem

修改 `RobotControlSystem.py`，分离 gRPC 启动和硬件初始化：

```python
class RobotControlSystem:
    def __init__(self, config: Dict[str, Any] = None, use_mock: bool = True, report: bool = True):
        # ... 现有初始化代码 ...

        # 新增：硬件管理器（延迟初始化）
        self.hardware_manager = None
        self.task_manager = None

        # 新增：硬件初始化状态
        self._hardware_initialized = False

    def start(self):
        """启动机器人控制系统（仅启动 gRPC 通信）"""
        if self.is_running:
            self.logger.warning("机器人控制系统已在运行")
            return

        self.logger.info("启动机器人控制系统...")

        # 1. 仅初始化 gRPC 客户端
        if not self._init_grpc_client():
            self.logger.error("gRPC 客户端初始化失败")
            return

        # 2. 设置命令处理器
        self.set_client_upload_response_handler(self._handle_clientUpload_response)
        self.set_server_command_response_handler(self._handle_serverCommand)

        # 3. 初始化硬件管理器（但不初始化硬件）
        self.hardware_manager = HardwareManager(self.config)
        self.hardware_manager.callbacks["on_component_state_change"].append(
            self._handle_component_state_change
        )

        self.is_running = True
        self.logger.info("机器人控制系统已启动（等待系统初始化命令）")

    def _init_hardware(self, component: str = "all", params: Dict = None) -> bool:
        """
        初始化硬件组件（由上位机命令触发）

        Args:
            component: 组件名称
            params: 初始化参数

        Returns:
            bool: 初始化是否成功
        """
        if not self.hardware_manager:
            self.logger.error("硬件管理器未初始化")
            return False

        # 初始化指定组件
        if not self.hardware_manager.init_component(component, params):
            return False

        # 如果是初始化全部组件，同时初始化 TaskManager
        if component == "all" or self.hardware_manager.is_all_ready():
            self._init_task_manager()
            self._hardware_initialized = True

            # 启动定时上报
            if self.report:
                self._start_reporting()

        return True

    def _init_task_manager(self):
        """初始化任务管理器"""
        if self.task_manager is not None:
            self.logger.warning("任务管理器已初始化")
            return

        # 创建适配器，将 HardwareManager 包装为 RobotController 接口
        robot_controller = HardwareRobotAdapter(self.hardware_manager)

        # 初始化任务管理器（使用适配器）
        self.task_manager = TaskManager(
            self.config,
            use_mock=self.use_mock,
            robot_controller=robot_controller  # 新增参数
        )

        # 注册回调
        self.task_manager.register_system_callback("on_command_status_change",
                                                    self._handle_command_status_callback)
        self.task_manager.register_system_callback("on_task_progress",
                                                    self._handle_task_progress_callback)
        self.task_manager.register_system_callback("on_operation_result",
                                                    self._handle_operation_result_callback)

        self.logger.info("任务管理器初始化完成")

    def _handle_system_control_cmd(self, command_envelope: CommandEnvelope) -> bool:
        """
        处理系统控制命令

        Args:
            command_envelope: 命令信封

        Returns:
            bool: 处理是否成功
        """
        try:
            data = command_envelope.data_json.get('system_control_cmd', {})
            action = data.get('action')
            component = data.get('component', 'all')
            params = data.get('params', {})

            self.logger.info(f"处理系统控制命令: action={action}, component={component}")

            if action == 'init':
                return self._init_hardware(component, params)
            elif action == 'enable':
                return self.hardware_manager.enable_component(component)
            elif action == 'disable':
                return self.hardware_manager.disable_component(component)
            elif action == 'shutdown':
                return self.hardware_manager.shutdown_component(component)
            else:
                self.logger.error(f"未知系统控制动作: {action}")
                return False

        except Exception as e:
            self.logger.error(f"处理系统控制命令失败: {e}")
            return False

    def _handle_serverCommand(self, response):
        """处理接收到的命令（修改版）"""
        command_envelope = convert_server_message_to_command_envelope(response)

        # 新增：处理系统控制命令
        if command_envelope.cmd_type == CmdType.SYSTEM_CONTROL_CMD:
            success = self._handle_system_control_cmd(command_envelope)
            # 发送命令执行结果
            self._send_system_control_response(command_envelope, success)
            return

        # 检查硬件是否已初始化（非系统控制命令需要硬件就绪）
        if not self._hardware_initialized:
            self.logger.warning(f"硬件未初始化，无法处理命令: {command_envelope.cmd_type}")
            # 可以发送错误响应
            return
```

### 2.4 修改 TaskManager

修改 `task/TaskManager.py`，支持外部传入 robot_controller：

```python
class TaskManager:
    def __init__(self, config: Dict[str, Any] = None, use_mock: bool = True,
                 robot_controller=None):
        """
        初始化任务管理器

        Args:
            config: 系统配置字典
            use_mock: 是否使用Mock机器人控制器
```

```
   ↓

   ↓
```
