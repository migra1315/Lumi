- RobotController.capture() 函数修改
- RTMP推流实现

```python
def _init_camera(self) -> bool:
    """初始化Orbbec相机"""
    try:
        from pyorbbecsdk import Pipeline, Config, OBSensorType, OBFormat, OBStreamType

        # 创建管道
        self._pipeline = Pipeline()
        self._orbbec_config = Config()

        # 配置RGB流
        try:
            profile_list = self._pipeline.get_stream_profile_list(OBSensorType.COLOR_SENSOR)
            # 尝试获取指定分辨率的配置
            rgb_profile = None
            for i in range(profile_list.count()):
                profile = profile_list.get_video_stream_profile(i)
                if (profile.get_width() == self.resolution['width'] and
                    profile.get_height() == self.resolution['height']):
                    rgb_profile = profile
                    break

            if rgb_profile is None:
                # 使用默认配置
                rgb_profile = profile_list.get_default_video_stream_profile()
                self.logger.warning(f"未找到指定分辨率，使用默认: {rgb_profile.get_width()}x{rgb_profile.get_height()}")

            self._orbbec_config.enable_stream(rgb_profile)

        except Exception as e:
            self.logger.warning(f"无法获取RGB流配置: {e}")
            self._orbbec_config.enable_stream(OBStreamType.COLOR, 0, 0, OBFormat.RGB)

        # 启动管道
        self._pipeline.start(self._orbbec_config)

        self._set_state(CameraState.CONNECTED)
        self.logger.info(f"相机初始化成功: {self.resolution['width']}x{self.resolution['height']}")
        return True

    except Exception as e:
        self.logger.error(f"相机初始化失败: {e}")
        self._set_state(CameraState.ERROR)
        return False
```

#### 3.3.2 帧采集循环

```python
def _capture_loop(self):
    """帧采集循环（独立线程）"""
    from pyorbbecsdk import OBFormat

    self.logger.info("帧采集线程已启动")
    frame_count = 0
    last_fps_time = time.time()

    while not self._stop_capture:
        try:
            # 等待帧（100ms超时）
            frames = self._pipeline.wait_for_frames(100)
            if frames is None:
                continue

            color_frame = frames.get_color_frame()
            if color_frame is None:
                continue

            # 转换为numpy数组
            color_data = color_frame.get_data()
            width = color_frame.get_width()
            height = color_frame.get_height()
            format_type = color_frame.get_format()

            if format_type == OBFormat.RGB:
                frame = np.frombuffer(color_data, dtype=np.uint8)
                frame = frame.reshape((height, width, 3))
                # RGB转BGR（OpenCV格式）
                frame = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
            elif format_type == OBFormat.MJPG:
                frame = cv2.imdecode(np.frombuffer(color_data, dtype=np.uint8), cv2.IMREAD_COLOR)
                if frame is None:
                    continue
            else:
                continue

            # 更新帧缓存（线程安全）
            with self._frame_lock:
                self._frame_buffer = frame
                self._frame_timestamp = time.time()

            self._stats['frames_captured'] += 1
            frame_count += 1

            # 计算实际FPS
            current_time = time.time()
            if current_time - last_fps_time >= 1.0:
                self._stats['fps_actual'] = frame_count / (current_time - last_fps_time)
                frame_count = 0
                last_fps_time = current_time

        except Exception as e:
            self._stats['capture_errors'] += 1
            self.logger.error(f"帧采集异常: {e}")
            time.sleep(0.1)

    self.logger.info("帧采集线程已退出")
```

#### 3.3.3 帧获取（抓拍）

```python
def capture_frame(self, timeout: float = 1.0) -> Optional[np.ndarray]:
    """
    获取当前帧

    Args:
        timeout: 等待超时时间（秒）

    Returns:
        numpy数组格式的图像，失败返回None
    """
    start_time = time.time()

    while time.time() - start_time < timeout:
        with self._frame_lock:
            if self._frame_buffer is not None:
                # 返回帧的副本，避免外部修改
                return self._frame_buffer.copy()
        time.sleep(0.01)

    self.logger.warning("获取帧超时")
    return None

def capture_to_base64(self, quality: int = 95) -> Optional[str]:
    """
    获取当前帧并转换为Base64

    Args:
        quality: JPEG压缩质量 (1-100)

    Returns:
        Base64编码的图像字符串，失败返回None
    """
    frame = self.capture_frame()
    if frame is None:
        return None

    try:
        # 编码为JPEG
        encode_param = [int(cv2.IMWRITE_JPEG_QUALITY), quality]
        _, buffer = cv2.imencode('.jpg', frame, encode_param)

        # 转换为Base64
        img_base64 = base64.b64encode(buffer).decode('utf-8')
        return img_base64

    except Exception as e:
        self.logger.error(f"图像编码失败: {e}")
        return None
```

#### 3.3.4 RTMP推流

```python
def _start_ffmpeg_stream(self) -> bool:
    """启动FFmpeg推流进程"""
    try:
        # 构建FFmpeg命令
        # 输入：从stdin读取原始帧数据
        # 输出：RTMP流
        width = self.resolution['width']
        height = self.resolution['height']
        fps = self.fps

        ffmpeg_cmd = [
            'ffmpeg',
            '-y',  # 覆盖输出
            '-f', 'rawvideo',
            '-vcodec', 'rawvideo',
            '-pix_fmt', 'bgr24',
            '-s', f'{width}x{height}',
            '-r', str(fps),
            '-i', '-',  # 从stdin读取
            '-c:v', 'libx264',
            '-preset', 'ultrafast',  # 最快编码，适合实时推流
            '-tune', 'zerolatency',  # 低延迟调优
            '-pix_fmt', 'yuv420p',
            '-g', str(fps * 2),  # GOP大小
            '-b:v', self.stream_config.get('bitrate', '2000k'),
            '-maxrate', self.stream_config.get('maxrate', '2500k'),
            '-bufsize', self.stream_config.get('bufsize', '5000k'),
            '-f', 'flv',
            self.rtmp_url
        ]

        self.logger.info(f"启动FFmpeg推流: {self.rtmp_url}")

        self._ffmpeg_process = subprocess.Popen(
            ffmpeg_cmd,
            stdin=subprocess.PIPE,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.PIPE
        )

        return True

    except Exception as e:
        self.logger.error(f"启动FFmpeg失败: {e}")
        return False

def _stream_loop(self):
    """推流循环（独立线程）"""
    self.logger.info("推流线程已启动")

    frame_interval = 1.0 / self.fps
    last_frame_time = time.time()

    while not self._stop_stream:
        try:
            # 控制帧率
            current_time = time.time()
            elapsed = current_time - last_frame_time
            if elapsed < frame_interval:
                time.sleep(frame_interval - elapsed)

            # 获取当前帧
            with self._frame_lock:
                if self._frame_buffer is None:
                    continue
                frame = self._frame_buffer

            # 写入FFmpeg stdin
            if self._ffmpeg_process and self._ffmpeg_process.poll() is None:
                try:
                    self._ffmpeg_process.stdin.write(frame.tobytes())
                    self._stats['frames_streamed'] += 1
                except BrokenPipeError:
                    self.logger.error("FFmpeg进程意外退出")
                    self._restart_ffmpeg()

            last_frame_time = time.time()

        except Exception as e:
            self._stats['stream_errors'] += 1
            self.logger.error(f"推流异常: {e}")
            time.sleep(0.1)

    self.logger.info("推流线程已退出")

def _restart_ffmpeg(self):
    """重启FFmpeg进程"""
    self.logger.warning("正在重启FFmpeg...")
    self._stop_ffmpeg()
    time.sleep(1)
    self._start_ffmpeg_stream()
```

```python
def capture(self, device_id: str) -> Dict[str, Any]:
    """
    拍照操作 - 从真实相机采集图像并返回base64编码数据

    Args:
        device_id: 设备ID

    Returns:
        Dict: {
            'success': bool,
            'images': List[str],  # base64图像列表
            'message': str,
            'device_id': str,
            'timestamp': float,
            'duration': float
        }
    """
    try:
        self.logger.info(f"执行拍照操作 - 设备ID: {device_id}")
        start_time = time.time()

        # 检查相机管理器是否可用
        if self.camera_manager is None:
            self.logger.warning("相机管理器未初始化，返回模拟数据")
            import base64
            mock_image_data = b"mock_image_bytes"
            img_base64 = base64.b64encode(mock_image_data).decode('utf-8')
            return {
                'success': True,
                'images': [img_base64],
                'message': f'模拟拍照成功',
                'device_id': device_id,
                'timestamp': time.time(),
                'duration': time.time() - start_time
            }

        # 从相机获取图像
        images = []

        # 拍摄多张图像（默认2张，可配置）
        num_captures = self.system_config.get('capture_count', 2)
        capture_interval = self.system_config.get('capture_interval', 0.5)  # 两次拍摄间隔

        for i in range(num_captures):
            img_base64 = self.camera_manager.capture_to_base64(quality=95)
            if img_base64:
                images.append(img_base64)
                self.logger.debug(f"第{i+1}张图像采集成功")
            else:
                self.logger.warning(f"第{i+1}张图像采集失败")

            if i < num_captures - 1:
                time.sleep(capture_interval)

        duration = time.time() - start_time
        success = len(images) > 0

        return {
            'success': success,
            'images': images,
            'message': f'拍照{"成功" if success else "失败"}，采集{len(images)}/{num_captures}张，耗时{duration:.2f}秒',
            'device_id': device_id,
            'timestamp': time.time(),
            'duration': duration
        }

    except Exception as e:
        self.logger.error(f"拍照失败: {e}")
        return {
            'success': False,
            'images': [],
            'message': f'拍照失败: {str(e)}',
            'device_id': device_id,
            'timestamp': time.time(),
            'duration': time.time() - start_time if 'start_time' in locals() else 0.0
        }
```

## 6. 推流与抓拍冲突解决方案

### 6.1 冲突分析

| 场景 | 冲突点 | 影响 |
|------|--------|------|
| 同时读取帧 | 帧数据竞争 | 可能导致帧丢失或数据不一致 |
| 抓拍期间推流 | CPU/带宽竞争 | 可能导致推流卡顿 |
| 推流进程占用相机 | 设备独占 | 无法抓拍 |

### 6.2 解决方案：帧共享模式

**核心设计**：采集线程独立运行，抓拍和推流都从帧缓冲区读取，不直接操作相机。

```
┌─────────────────────────────────────────────────────────┐
│                    Capture Thread                        │
│   (唯一的相机访问者，持续采集帧到缓冲区)                    │
└─────────────────────┬───────────────────────────────────┘
                      │
                      ▼
┌─────────────────────────────────────────────────────────┐
│                   Frame Buffer                           │
│   ┌─────────────────────────────────────────────────┐   │
│   │  RLock保护的共享帧（numpy数组）                    │   │
│   │  - 写入：capture_thread (独占)                    │   │
│   │  - 读取：stream_thread, capture_to_base64 (共享) │   │
│   └─────────────────────────────────────────────────┘   │
└────────────┬────────────────────────────┬───────────────┘
             │                            │
             ▼                            ▼
┌────────────────────────┐  ┌────────────────────────────┐
│    Stream Thread       │  │    capture_to_base64()     │
│  (读取帧→FFmpeg推流)    │  │  (读取帧→编码→Base64)      │
└────────────────────────┘  └────────────────────────────┘
```

### 6.3 实现细节

#### 6.3.1 使用RLock实现读写锁语义

```python
class CameraManager:
    def __init__(self):
        # 使用RLock支持递归锁定
        self._frame_lock = threading.RLock()
        self._frame_buffer = None
        self._frame_timestamp = 0

    def _update_frame(self, frame: np.ndarray):
        """更新帧缓冲（写操作）"""
        with self._frame_lock:
            self._frame_buffer = frame
            self._frame_timestamp = time.time()

    def _read_frame(self) -> Optional[np.ndarray]:
        """读取帧缓冲（读操作，返回副本）"""
        with self._frame_lock:
            if self._frame_buffer is not None:
                return self._frame_buffer.copy()
            return None
```

#### 6.3.2 抓拍优先级

```python
def capture_to_base64(self, quality: int = 95, priority: bool = True) -> Optional[str]:
    """
    获取当前帧并转换为Base64

    Args:
        quality: JPEG压缩质量
        priority: 是否使用高优先级（暂停推流）
    """
    if priority and self._is_streaming:
        # 临时暂停推流线程，确保抓拍获得最新帧
        self._pause_streaming()

    try:
        # 等待新帧到达
        old_timestamp = self._frame_timestamp
        timeout = 0.5
        start = time.time()

        while time.time() - start < timeout:
            if self._frame_timestamp > old_timestamp:
                break
            time.sleep(0.01)

        # 获取帧并编码
        frame = self._read_frame()
        if frame is None:
            return None

        encode_param = [int(cv2.IMWRITE_JPEG_QUALITY), quality]
        _, buffer = cv2.imencode('.jpg', frame, encode_param)
        return base64.b64encode(buffer).decode('utf-8')

    finally:
        if priority and self._is_streaming:
            self._resume_streaming()
```

### 6.4 性能优化建议

1. **帧率控制**：推流帧率可适当低于采集帧率，减少CPU占用
2. **分辨率调整**：推流可使用较低分辨率，抓拍使用全分辨率
3. **异步编码**：抓拍编码可在独立线程中进行
4. **缓冲区策略**：使用固定大小的环形缓冲区，避免内存溢出

## 11. 风险与缓解

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
